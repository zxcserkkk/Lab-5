## Реализация программы

```c
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <math.h>
#include <stdlib.h>
#include <locale.h>

int main(void)
{
    setlocale(LC_ALL, ".UTF-8");

    const double x = 17.421;
    const double y = 10.365e-3;
    const double z = 0.828e5;
    double f;

    f = pow(y + cbrt(x - 1), 1.0 / 4) /
        (fabs(x - y) * (pow(sin(z), 2) + tan(z)));

    printf("Исходные данные:\n");
    printf("x = %g\n", x);
    printf("y = %g\n", y);
    printf("z = %g\n", z);
    printf("Результат: f = %.5f\n", f);

    system("pause");
    return 0;
}
```
